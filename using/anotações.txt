    /*
    * uv_tcp_t is a subclass of uv_stream_t.
    *
    * Represents a TCP stream or TCP server.
    */
    struct uv_tcp_s {
        UV_HANDLE_FIELDS
        UV_STREAM_FIELDS
        UV_TCP_PRIVATE_FIELDS
    };
